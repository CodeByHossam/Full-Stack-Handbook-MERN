Phaser: Physics, Collisions & Groups

Arcade physics, static platforms, collision detection, groups.

Physics Config

const config = {
  type: Phaser.AUTO, width:800,height:600,
  physics: { default: 'arcade', arcade: { gravity: { y: 300 }, debug: false } },
  scene: { preload, create, update }
};

Platforms & Collisions

function create() {
  this.platforms = this.physics.add.staticGroup();
  this.platforms.create(400, 568, 'ground').setScale(2).refreshBody();
  this.player = this.physics.add.sprite(100, 450, 'player');
  this.physics.add.collider(this.player, this.platforms);
}

Groups & Overlap

this.coins = this.physics.add.group({ key:'coin', repeat:11, setXY:{x:12,y:0,stepX:70} });
this.physics.add.collider(this.coins, this.platforms);
this.physics.add.overlap(this.player, this.coins, (p,c)=>{ c.disableBody(true,true); }, null, this);
